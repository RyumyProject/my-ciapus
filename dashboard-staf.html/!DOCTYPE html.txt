    <script>
        // Set nama profil dari session login
        document.getElementById('namaUser').innerText = localStorage.getItem('user_nama') || "User";
        document.getElementById('roleUser').innerText = localStorage.getItem('user_role') || "Staf";

        // Jam Real-time
        setInterval(() => {
            document.getElementById('jamLive').innerText = new Date().toLocaleTimeString() + " WIB";
        }, 1000);

        function prosesAbsen(tipe) {
            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(function(position) {
                    let lat = position.coords.latitude;
                    let lon = position.coords.longitude;
                    
                    // Simpan data absen ke Firebase
                    db.collection("absensi").add({
                        nama: localStorage.getItem('user_nama'),
                        role: localStorage.getItem('user_role'),
                        waktu: new Date().toLocaleString(),
                        tipe: tipe,
                        lokasi: `${lat}, ${lon}`,
                        status: "Hadir"
                    }).then(() => {
                        alert(`Absen ${tipe} Berhasil dicatat dengan GPS!`);
                    });
                }, function() {
                    alert("Gagal mengambil GPS. Pastikan izin lokasi aktif.");
                });
            } else {
                alert("Browser tidak mendukung GPS.");
            }
        }

        function logout() {
            localStorage.clear();
            window.location.href = "index.html";
        }
    </script>
